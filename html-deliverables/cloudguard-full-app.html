Create IAM Role in AWS

Run this Terraform snippet in your AWS account. It creates a read-only role CloudGuard uses to read billing data.
# Save as cloudguard-role.tf, then: terraform init && terraform apply resource "aws_iam_role" "cloudguard_readonly" { name = "CloudGuardReadOnly" assume_role_policy = jsonencode({ Statement = [{ Action = "sts:AssumeRole" Effect = "Allow" Principal = { AWS = "arn:aws:iam::YOUR_CG_ACCOUNT:role/collector" } Condition = { StringEquals = { "sts:ExternalId" = "cg-8f3a2b91-4e7d-49ac-b2c5-d7e1f0a83924" } } }] }) } resource "aws_iam_role_policy" "cloudguard" { role = aws_iam_role.cloudguard_readonly.id policy = jsonencode({ Statement = [{ Effect = "Allow" Action = ["ce:GetCostAndUsage","ce:GetCostForecast", "cloudwatch:GetMetricData","ec2:DescribeInstances", "ec2:DescribeVolumes","rds:DescribeDBInstances"] Resource = "*" }] }) } output "role_arn" { value = aws_iam_role.cloudguard_readonly.arn }
