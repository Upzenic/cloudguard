System architecture
Serverless-first on AWS. You already know this stack. Zero new tools to learn.
Design principle: Everything runs in YOUR AWS account. Customers grant read-only cross-account IAM access. You never touch their infra — you only read billing + CloudWatch metrics.

External ID pattern: Generate a UUID per customer when they onboard. Store it in your DB. Pass it in AssumeRole calls. This prevents confused deputy attacks — a critical security requirement.

Data collector Lambda — main logic

lambdas/data_collector/handler.py
python 
import boto3
import json
from datetime import datetime, timedelta
from app.models import Customer, CostRecord
from app.core.db import get_db

def assume_customer_role(customer: Customer):
    """Assume the cross-account IAM role for a customer."""
    sts = boto3.client('sts')
    response = sts.assume_role(
        RoleArn=f"arn:aws:iam::{customer.aws_account_id}:role/CloudGuardReadOnly",
        RoleSessionName=f"cloudguard-{customer.id}",
        ExternalId=customer.external_id,  # UUID stored in your DB
        DurationSeconds=3600
    )
    creds = response['Credentials']
    return {
        'aws_access_key_id':     creds['AccessKeyId'],
        'aws_secret_access_key': creds['SecretAccessKey'],
        'aws_session_token':     creds['SessionToken']
    }

def get_monthly_costs(credentials: dict, months_back: int = 3):
    """Pull last N months of cost data via Cost Explorer."""
    ce = boto3.client('ce', region_name='us-east-1', **credentials)
    end   = datetime.now().replace(day=1).strftime('%Y-%m-%d')
    start = (datetime.now() - timedelta(days=90)).replace(day=1).strftime('%Y-%m-%d')

    response = ce.get_cost_and_usage(
        TimePeriod={'Start': start, 'End': end},
        Granularity='MONTHLY',
        Metrics=['UnblendedCost'],
        GroupBy=[{'Type': 'DIMENSION', 'Key': 'SERVICE'}]
    )
    return response['ResultsByTime']

def find_idle_resources(credentials: dict):
    """Find EC2 instances with <5% CPU over last 7 days."""
    ec2 = boto3.client('ec2', **credentials)
    cw  = boto3.client('cloudwatch', **credentials)

    instances = ec2.describe_instances(
        Filters=[{'Name': 'instance-state-name', 'Values': ['running']}]
    )

    idle = []
    for reservation in instances['Reservations']:
        for instance in reservation['Instances']:
            metrics = cw.get_metric_data(
                MetricDataQueries=[{
                    'Id': 'cpu',
                    'MetricStat': {
                        'Metric': {
                            'Namespace': 'AWS/EC2',
                            'MetricName': 'CPUUtilization',
                            'Dimensions': [{
                                'Name': 'InstanceId',
                                'Value': instance['InstanceId']
                            }]
                        },
                        'Period': 86400,  # daily average
                        'Stat': 'Average'
                    }
                }],
                StartTime=datetime.now() - timedelta(days=7),
                EndTime=datetime.now()
            )
            avg_cpu = sum(metrics['MetricDataResults'][0]['Values']) / 7
            if avg_cpu < 5.0:  # Less than 5% CPU = idle
                idle.append({
                    'instance_id': instance['InstanceId'],
                    'type':        instance['InstanceType'],
                    'avg_cpu':     round(avg_cpu, 2),
                    'monthly_cost': estimate_ec2_cost(instance['InstanceType'])
                })
    return idle

def lambda_handler(event, context):
    """EventBridge triggers this daily at 06:00 IST."""
    db = get_db()
    customers = db.query(Customer).filter(Customer.active == True).all()

    for customer in customers:
        try:
            creds  = assume_customer_role(customer)
            costs  = get_monthly_costs(creds)
            idle   = find_idle_resources(creds)
            save_to_db(customer.id, costs, idle)
            check_and_send_alerts(customer, costs)
        except Exception as e:
            # Don't let one customer failure break others
            print(f"Error for customer {customer.id}: {e}")
            continue

    return {'statusCode': 200, 'body': f'Processed {len(customers)} customers'}

import boto3
from twilio.rest import Client as TwilioClient

ses     = boto3.client('ses', region_name='ap-south-1')   # Mumbai
twilio  = TwilioClient(TWILIO_SID, TWILIO_AUTH)

def check_and_send_alerts(customer, costs):
    current_month_spend = get_current_month_total(costs)
    budget              = customer.monthly_budget_inr

    # Alert at 80% of budget
    if current_month_spend >= (budget * 0.8):
        pct = int((current_month_spend / budget) * 100)
        msg = (
            f"☁️ CloudGuard Alert\n"
            f"Account: {customer.company_name}\n"
            f"Spend: ₹{current_month_spend:,.0f} ({pct}% of budget)\n"
            f"Projected month-end: ₹{project_month_end(costs):,.0f}\n"
            f"View dashboard: https://app.cloudguard.in/dashboard"
        )
        # WhatsApp via Twilio
        if customer.whatsapp_number:
            twilio.messages.create(
                body=msg,
                from_='whatsapp:+14155238886',   # Twilio sandbox number
                to=f'whatsapp:{customer.whatsapp_number}'
            )
        # Email via SES
        ses.send_email(
            Source='alerts@cloudguard.in',
            Destination={'ToAddresses': [customer.email]},
            Message={
                'Subject': {'Data': f'⚠️ {pct}% of your AWS budget used'},
                'Body':    {'Text': {'Data': msg}}
            }
        )
EventBridge rule: cron(30 0 * * ? *) — runs at 06:00 IST (00:30 UTC) daily. Attach to your data_collector Lambda. Cost: essentially zero (Lambda free tier covers thousands of daily invocations).

Stripe subscription flow

app/services/stripe_service.py
python 
import stripe
from fastapi import APIRouter, Request, HTTPException

stripe.api_key = STRIPE_SECRET_KEY
router = APIRouter()

# ── 1. Create checkout session (called when user clicks "Start Trial") ──
async def create_checkout_session(customer_email: str, plan: str):
    PRICE_IDS = {
        "starter":  "price_xxx",   # ₹2,999/mo — create in Stripe dashboard
        "growth":   "price_yyy",   # ₹5,999/mo
        "business": "price_zzz",   # ₹12,999/mo
    }
    session = stripe.checkout.Session.create(
        payment_method_types=['card'],
        mode='subscription',
        customer_email=customer_email,
        line_items=[{'price': PRICE_IDS[plan], 'quantity': 1}],
        subscription_data={
            'trial_period_days': 30  # Free trial — card stored, not charged
        },
        success_url='https://app.cloudguard.in/onboarding?session={CHECKOUT_SESSION_ID}',
        cancel_url='https://cloudguard.in/pricing',
    )
    return session.url  # Redirect customer to this Stripe-hosted page

# ── 2. Webhook — Stripe calls THIS when payment events happen ──
@router.post("/webhooks/stripe")
async def stripe_webhook(request: Request):
    payload   = await request.body()
    sig       = request.headers.get('stripe-signature')

    try:
        event = stripe.Webhook.construct_event(payload, sig, STRIPE_WEBHOOK_SECRET)
    except ValueError:
        raise HTTPException(status_code=400)

    if event['type'] == 'customer.subscription.created':
        activate_customer(event['data']['object'])

    elif event['type'] == 'invoice.payment_succeeded':
        extend_subscription(event['data']['object'])

    elif event['type'] == 'customer.subscription.deleted':
        deactivate_customer(event['data']['object'])  # stop data collection

    elif event['type'] == 'invoice.payment_failed':
        send_payment_failed_email(event['data']['object'])

    return {"status": "ok"}
Stripe India note: Stripe in India requires a registered business entity (proprietorship, LLP, or Pvt Ltd) for INR payouts. Start with Razorpay — instant activation with just PAN + bank account. Add Stripe later for international customers.

Razorpay (India) — subscription setup

app/services/razorpay_service.py
python 
import razorpay
import hmac, hashlib

client = razorpay.Client(auth=(RAZORPAY_KEY_ID, RAZORPAY_KEY_SECRET))

def create_subscription(customer_email: str, plan_id: str):
    """
    plan_id = Razorpay Plan ID created in dashboard
    plan_starter  → ₹2,999/month
    plan_growth   → ₹5,999/month
    plan_business → ₹12,999/month
    """
    subscription = client.subscription.create({
        "plan_id":         plan_id,
        "total_count":     120,     # 10 years — effectively infinite
        "quantity":        1,
        "customer_notify": 1,
        "notes": {"email": customer_email}
    })
    return subscription["short_url"]  # Payment link, send to customer

def verify_webhook(body: bytes, signature: str):
    """Always verify Razorpay webhook signatures."""
    expected = hmac.new(
        RAZORPAY_WEBHOOK_SECRET.encode(),
        body, hashlib.sha256
    ).hexdigest()
    return hmac.compare_digest(expected, signature)

GitHub Actions — prod deploy

.github/workflows/deploy-prod.yml
yaml 
name: Deploy to Production

on:
  push:
    branches: [main]

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Run tests
        run: |
          pip install -r backend/requirements.txt
          pytest backend/tests/ -v
          # Tests must pass before ANY deploy

  build-and-push:
    needs: test
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          aws-access-key-id:     ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region:            ap-south-1  # Mumbai

      - name: Login to ECR
        uses: aws-actions/amazon-ecr-login@v2

      - name: Build and push Docker image
        env:
          ECR_REGISTRY:   ${{ steps.login-ecr.outputs.registry }}
          IMAGE_TAG:      ${{ github.sha }}
        run: |
          docker build -t $ECR_REGISTRY/cloudguard-backend:$IMAGE_TAG ./backend
          docker push $ECR_REGISTRY/cloudguard-backend:$IMAGE_TAG
          echo "IMAGE=$ECR_REGISTRY/cloudguard-backend:$IMAGE_TAG" >> $GITHUB_ENV

  deploy:
    needs: build-and-push
    runs-on: ubuntu-latest
    environment: production   # Requires manual approval in GitHub
    steps:
      - name: Deploy to ECS
        run: |
          aws ecs update-service \
            --cluster cloudguard-prod \
            --service cloudguard-backend \
            --force-new-deployment \
            --region ap-south-1

      - name: Deploy frontend to S3 + invalidate CloudFront
        run: |
          cd frontend && npm ci && npm run build
          aws s3 sync dist/ s3://cloudguard-app-prod --delete
          aws cloudfront create-invalidation \
            --distribution-id ${{ secrets.CF_DIST_ID }} \
            --paths "/*"

      - name: Update Lambda
        run: |
          cd backend/lambdas/data_collector
          zip -r function.zip .
          aws lambda update-function-code \
            --function-name cloudguard-data-collector \
            --zip-file fileb://function.zip

Secrets management: All secrets (DB password, Stripe key, Twilio SID, Razorpay key) go into AWS Secrets Manager. Lambda and ECS fetch them at runtime via boto3.client('secretsmanager'). Never hardcode. Never in environment variables in plain text.
